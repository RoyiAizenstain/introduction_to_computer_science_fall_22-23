system_capacity = 30

                ("second job", 2, 5),
                ("third job", 5, 5),
                ("forth job", 4, 16),
                ("fifth job", 1, 3),
                ("sixth job", 3, 5),
                ("seventh job", 6, 12),
                ("eight", 2, 4),
                ("nine", 9, 3),
                ("extra", 6, 1),
                ("nine", 1, 15),
                ("",), ]
job_req_sys = JobRequestSystem(nodes_num, system_capacity)
for element in job_requests:
    job_instance = job_req_sys.request_new_job(*element)  # (*request_job_window())
    print()
    print("*" * 10 + "Requesting the job:" + "*" * 10)
    print(job_instance)
    job_req_sys.request_job_resources(job_instance)
    print("Nodes situation:")
    print(job_req_sys.get_nodes_system_capacity())
def print_curr_situtaion(jrs):
    print("\nCurrent situation:") 
    print("#" * 38)
    print("Initialized jobs:")
    print(jrs.get_initialized_jobs())
    print("#" * 38)
    print("Waiting jobs:")
    print(jrs.get_waiting_jobs())
    print("#" * 38)
    print("Nodes situation:")
    print(jrs.get_nodes_system_capacity())
    
print("\n"+ "%"*25 + "-"* 10 + "%" *25+ "\n")
